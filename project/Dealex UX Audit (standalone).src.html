         'M4.22 19.78l1.42-1.42M18.36 5.64l1.42-1.42',
  // …add the rest as you migrate
}

export function Icon({ name, size = 18, className = '', strokeWidth = 1.6 }) {
  const d = PATHS[name]
  if (!d) return null
  return (
    <svg width={size} height={size} viewBox="0 0 24 24" fill="none"
         stroke="currentColor" strokeWidth={strokeWidth}
         strokeLinecap="round" strokeLinejoin="round"
         className={`icon ${className}`} aria-hidden="true">
      <path d={d} />
    </svg>
  )
}
06 / Day-one quick wins

What to ship by end of week

If you do nothing else from this audit, do these. Each one is under an afternoon's work and lifts perceived quality more than its size would suggest.
~ 30 min
Retire emoji from nav, theme toggle & role cards
Drop the Icon.jsx from Patch C. Swap 🏠 🏦 📄 💼 👨‍💼 🌙 🏛️ 📜 🤝 for the named SVG glyphs. Single biggest perceived-quality jump on the list.
~ 20 min
Rename --gold → --accent
One sed across src/. Removes the cognitive mismatch every new dev (and you, six months from now) will hit on day one.
~ 1 hour
Bump every type size below 13px
Set :root { font-size: 14px } as the floor, then triage the 7–10px hardcoded values in Landing.jsx. Body becomes readable on a phone for the first time.
~ 2 hours
Lead the hero with a CTA + a benefit
Move the "Ground Rules" callout below the role selector. Replace with a primary "Start a deal" + secondary "How it works." Re-frame "no Zoom" as "audited, sequenced rail."
~ 30 min
Re-label the ticker as indicative
Either wire a 5-min cached feed, or add · INDICATIVE · 12s ago and pulse a dot. Static "live" prices are the single most product-eroding detail on the screen.
~ 1 hour
Add a :focus-visible ring + 44px min tap target
Two global rules in global.css. Unlocks keyboard nav and passes the basic mobile-a11y sweep. No design decisions required.
